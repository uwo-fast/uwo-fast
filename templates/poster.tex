% #1 width, #2 height, #3 image path, #4 credit (may be empty). Scales to
% fill the box and crops the overflow, so photos of any aspect ratio line up.
% The credit sits on the photo's corner so it costs the layout no height.
\newsavebox\imgbox
\newcommand{\fillimage}[4]{%
  \sbox\imgbox{\includegraphics[width=#1]{#3}}%
  \ifdim\ht\imgbox<#2\relax\sbox\imgbox{\includegraphics[height=#2]{#3}}\fi
  \begin{tikzpicture}
    \clip (0,0) rectangle (#1,#2);
    \node[inner sep=0pt] at (#1/2,#2/2) {\usebox\imgbox};
    \if\relax\detokenize{#4}\relax\else
      \node[anchor=south east,fill=black,fill opacity=0.55,text opacity=1,
            text=white,font=\fontsize{6.5}{8}\selectfont,inner xsep=5pt,inner ysep=2.5pt]
        at (#1,0) {#4};
    \fi
  \end{tikzpicture}}
% A photo covering the box at (#1,#2) sized #3 x #4 (inches), for a panel
% whose clip is already set.
\newcommand{\panelimage}[5]{%
  \sbox\imgbox{\includegraphics[width=#3in]{#5}}%
  \ifdim\ht\imgbox<#4in\relax\sbox\imgbox{\includegraphics[height=#4in]{#5}}\fi
  \node[inner sep=0pt] at (#1+#3/2,#2+#4/2) {\usebox\imgbox};}
% Same box, marked as missing. Approved posters refuse to build with one.
\newcommand{\placeholder}[3]{%
  \begin{tikzpicture}
    \fill[tint] (0,0) rectangle (#1,#2);
    \draw[accent,dashed,line width=0.8pt] (0,0) rectangle (#1,#2);
    \node[text width=#1-0.5in,align=center,font=\small\color{accent}] at (#1/2,#2/2)
      {{\condensed\bfseries IMAGE NEEDED}\\[3pt]#3};
  \end{tikzpicture}}
\newcommand{\kicker}[1]{{\condensed\fontsize{9}{11}\selectfont\color{accent}\addfontfeature{LetterSpace=8}\MakeUppercase{#1}}}
